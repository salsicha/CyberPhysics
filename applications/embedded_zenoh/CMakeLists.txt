cmake_minimum_required(VERSION 3.20)
option(CP_PICO "Build Raspberry Pi Pico 2 W firmware" OFF)
set(PICOROS_SOURCE /opt/picoros CACHE PATH "Pinned Pico-ROS checkout")
set(Z_TRANSPORT_LEASE 10000 CACHE STRING "Match embedded router's bounded reconnect lease")
set(Z_TRANSPORT_LEASE_EXPIRE_FACTOR 2 CACHE STRING "Match native ROS router")
set(Z_FEATURE_LINK_SERIAL 1 CACHE STRING "Native Zenoh serial transport")
set(Z_FEATURE_AUTO_RECONNECT 1 CACHE STRING "Restore the session and ROS declarations after link loss")
set(PICOROS_BUILD_EXAMPLES OFF CACHE BOOL "")
set(PICOROS_BUILD_TESTS OFF CACHE BOOL "")
if(CP_PICO)
  set(PICO_BOARD pico2_w CACHE STRING "")
  include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
  include(${PICOROS_SOURCE}/thirdparty/zenoh-pico/examples/rpi_pico/FreeRTOS_Kernel_import.cmake)
endif()
project(cyberphysics_embedded_zenoh C CXX ASM)
set(CMAKE_C_STANDARD 11)
if(CP_PICO)
  pico_sdk_init()
  set(ZP_PLATFORM rpi_pico CACHE STRING "")
  set(Z_FEATURE_LINK_SERIAL_USB 1 CACHE STRING "")
  set(Z_FEATURE_UNSTABLE_API 1 CACHE STRING "")
  set(WIFI_SUPPORT_ENABLED 1)
  set(WIFI_SSID "" CACHE STRING "WiFi SSID; unused for USB")
  set(WIFI_PASSWORD "" CACHE STRING "WiFi password; unused for USB")
  set(ZENOH_CONFIG_MODE client)
  set(ZENOH_CONFIG_CONNECT "serial/usb#baudrate=115200" CACHE STRING "Zenoh router locator")
  set(ZENOH_CONFIG_LISTEN "")
  set(CP_ROS_DOMAIN_ID 0 CACHE STRING "ROS domain")
  set(PICO_EXAMPLES ${PICOROS_SOURCE}/thirdparty/zenoh-pico/examples/rpi_pico)
  configure_file(${PICO_EXAMPLES}/include/config.h.in ${CMAKE_CURRENT_BINARY_DIR}/config.h)
  add_compile_definitions(LWIP_TIMEVAL_PRIVATE=0 CP_ROS_DOMAIN_ID=${CP_ROS_DOMAIN_ID})
endif()
add_subdirectory(${PICOROS_SOURCE} picoros EXCLUDE_FROM_ALL)
if(CP_PICO)
  set(PICO_LIBS hardware_uart pico_stdlib pico_rand FreeRTOS-Kernel-Heap4
      pico_cyw43_arch_lwip_sys_freertos tinyusb_host tinyusb_device tinyusb_board)
  target_include_directories(zenohpico_static PRIVATE ${CMAKE_CURRENT_BINARY_DIR}
      ${PICO_EXAMPLES}/include ${PICO_EXAMPLES}/include/tusb)
  target_link_libraries(zenohpico_static PRIVATE ${PICO_LIBS})
  target_link_libraries(picoros ${PICO_LIBS})
  target_include_directories(picoros PRIVATE ${CMAKE_CURRENT_BINARY_DIR}
      ${PICO_EXAMPLES}/include ${PICO_EXAMPLES}/include/tusb)
  add_executable(embedded_zenoh src/demo.c src/pico.c ${PICO_EXAMPLES}/main.c)
  target_include_directories(embedded_zenoh PRIVATE ${CMAKE_CURRENT_BINARY_DIR}
      ${PICO_EXAMPLES}/include ${PICO_EXAMPLES}/include/tusb)
  target_link_libraries(embedded_zenoh PRIVATE picoros ${PICO_LIBS})
  pico_enable_stdio_usb(embedded_zenoh 0)
  pico_enable_stdio_uart(embedded_zenoh 1)
  pico_add_extra_outputs(embedded_zenoh)
else()
  add_executable(embedded_zenoh src/demo.c src/host.c)
  target_link_libraries(embedded_zenoh PRIVATE picoros)
endif()
target_include_directories(embedded_zenoh PRIVATE src)
set_source_files_properties(src/demo.c src/host.c src/pico.c
    PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
